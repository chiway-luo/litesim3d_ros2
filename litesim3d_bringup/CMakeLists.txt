cmake_minimum_required(VERSION 3.20)
project(litesim3d_bringup LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(litesim3d_runtime REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
find_package(OpenSSL REQUIRED)
add_library(model_preparation STATIC src/model.cpp src/controller_parameters.cpp src/xacro.cpp)
target_compile_features(model_preparation PUBLIC cxx_std_17)
target_compile_options(model_preparation PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(model_preparation PUBLIC litesim3d_runtime::litesim3d_runtime PkgConfig::TINYXML2 OpenSSL::Crypto)
ament_target_dependencies(model_preparation PUBLIC ament_index_cpp)
add_executable(prepare_config src/prepare_config.cpp)
target_link_libraries(prepare_config model_preparation)
add_executable(prepare_xacro src/prepare_xacro.cpp)
target_link_libraries(prepare_xacro model_preparation)
add_executable(make_rsp_params src/make_rsp_params.cpp)
target_link_libraries(make_rsp_params model_preparation)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
install(TARGETS prepare_config prepare_xacro make_rsp_params DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(native_model_unit test/test_native_model.cpp test/test_native_xacro.cpp)
  target_include_directories(native_model_unit PRIVATE src)
  target_link_libraries(native_model_unit model_preparation)
  # In-process orchestration units compile the real mains with provider spies;
  # no model pipeline, signal installation or child process is executed.
  ament_add_gtest(native_cli_unit test/test_native_cli_unit.cpp)
  target_include_directories(native_cli_unit PRIVATE src)
  target_link_libraries(native_cli_unit model_preparation)
  ament_add_pytest_test(model_preparation test APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1 TIMEOUT 60)
endif()
ament_package()
