if(APPLE)
  # The shell entrypoint drops inherited DYLD variables on macOS. Resolve this
  # project's generated ROS interfaces relative to the installed executable.
  get_filename_component(_camera_interfaces_prefix "${litesim3d_interfaces_DIR}/../../.." ABSOLUTE)
  set_property(TARGET camera_node APPEND PROPERTY INSTALL_RPATH
    "@loader_path/.." "@loader_path/../../../litesim3d_interfaces/lib" "${_camera_interfaces_prefix}/lib")
  set(LITESIM_METAL_PLUGIN "" CACHE FILEPATH "Project-local Ogre-Next Metal backend")
  if(LITESIM_METAL_PLUGIN)
    install(FILES "${LITESIM_METAL_PLUGIN}" DESTINATION lib/${PROJECT_NAME}/ogre2)
    # Ogre probes both backend names in this directory even in Metal mode.
    # Install the existing matching GL plugin alongside Metal to avoid a false
    # 'rendering will not be possible' message from that probe.
    find_package(PkgConfig REQUIRED)
    pkg_get_variable(_ogre_prefix OGRE-Next prefix)
    if(EXISTS "${_ogre_prefix}/lib/OGRE-Next/RenderSystem_GL3Plus.dylib")
      get_filename_component(_ogre_gl "${_ogre_prefix}/lib/OGRE-Next/RenderSystem_GL3Plus.dylib" REALPATH)
      install(FILES "${_ogre_gl}" DESTINATION lib/${PROJECT_NAME}/ogre2 RENAME RenderSystem_GL3Plus.dylib)
    endif()
    install(FILES licenses/LICENSE.OGRE DESTINATION share/${PROJECT_NAME})
  endif()
endif()
