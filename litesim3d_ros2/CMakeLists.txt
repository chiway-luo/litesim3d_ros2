cmake_minimum_required(VERSION 3.20)
project(litesim3d_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(litesim3d_core REQUIRED)
set(DEPS rclcpp geometry_msgs sensor_msgs nav_msgs rosgraph_msgs std_msgs std_srvs diagnostic_msgs visualization_msgs tf2_ros tf2_msgs litesim3d_interfaces)
foreach(dep ${DEPS})
  find_package(${dep} REQUIRED)
endforeach()
# Record the installed ROS/launch implementation independently of the core hash.
set(bridge_sources src/sim_node.cpp src/sim_world_node.cpp src/point_adapter.cpp
  src/point_adapter.hpp src/sim_ouster_adapter.cpp src/sim_cloud_probe.cpp
  launch/demo.launch.py launch/world.launch.py scripts/sim_session)
set(bridge_fingerprint "")
foreach(source ${bridge_sources})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND bridge_fingerprint "${source}:${source_hash}\n")
endforeach()
string(SHA256 bridge_hash "${bridge_fingerprint}")
add_compile_definitions(LWS_BRIDGE_SHA256="${bridge_hash}")
add_executable(sim_node src/sim_node.cpp)
target_link_libraries(sim_node litesim3d_core::litesim3d_core)
ament_target_dependencies(sim_node ${DEPS})
target_compile_options(sim_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sim_world_node src/sim_world_node.cpp)
target_link_libraries(sim_world_node litesim3d_core::litesim3d_core)
ament_target_dependencies(sim_world_node ${DEPS})
find_package(PCL REQUIRED COMPONENTS common filters)
add_executable(sim_ouster_adapter src/sim_ouster_adapter.cpp src/point_adapter.cpp)
ament_target_dependencies(sim_ouster_adapter rclcpp sensor_msgs std_msgs)
add_executable(sim_cloud_probe src/sim_cloud_probe.cpp)
ament_target_dependencies(sim_cloud_probe rclcpp sensor_msgs std_msgs)
target_include_directories(sim_cloud_probe SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(sim_cloud_probe ${PCL_LIBRARIES})
if(SIM_COVERAGE)
  foreach(target sim_ouster_adapter sim_cloud_probe)
    target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(${target} PRIVATE -fprofile-instr-generate)
  endforeach()
endif()
install(TARGETS sim_cloud_probe sim_ouster_adapter sim_world_node sim_node DESTINATION lib/${PROJECT_NAME})
find_package(Python3 REQUIRED COMPONENTS Interpreter)
configure_file(scripts/sim_session sim_session @ONLY)
install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/sim_session DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
ament_package()
