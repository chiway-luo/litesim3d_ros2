cmake_minimum_required(VERSION 3.20)
project(litesim3d_plugin_tests LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
if(BUILD_TESTING)
  find_package(litesim3d_plugin_api REQUIRED)
  find_package(litesim3d_core REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  find_package(GTest REQUIRED)
  find_package(yaml-cpp REQUIRED)
  find_package(pluginlib REQUIRED)
  add_library(contract_fault_plugins SHARED contract_fault_plugins.cpp)
  target_compile_features(contract_fault_plugins PRIVATE cxx_std_17)
  target_link_libraries(contract_fault_plugins litesim3d_core::litesim3d_core)
  ament_target_dependencies(contract_fault_plugins pluginlib)
  pluginlib_export_plugin_description_file(litesim3d_plugin_api contract_fault_plugins.xml)
  install(TARGETS contract_fault_plugins LIBRARY DESTINATION lib)
  add_executable(plugin_contract_tests test_motion.cpp)
  add_executable(plugin_odometry_probe probe_odometry.cpp)
  target_compile_features(plugin_odometry_probe PRIVATE cxx_std_17)
  target_link_libraries(plugin_odometry_probe litesim3d_plugin_api::litesim3d_plugin_api)
  install(TARGETS plugin_odometry_probe DESTINATION lib/${PROJECT_NAME})
  target_compile_features(plugin_contract_tests PRIVATE cxx_std_17)
  target_link_libraries(plugin_contract_tests litesim3d_plugin_api::litesim3d_plugin_api GTest::gtest_main yaml-cpp)
  ament_target_dependencies(plugin_contract_tests ament_index_cpp)
  enable_testing()
  include(GoogleTest)
  gtest_discover_tests(plugin_contract_tests DISCOVERY_TIMEOUT 60)
  add_executable(core_plugin_contract_tests test_core_plugins.cpp test_sensors.cpp test_host_contracts.cpp)
  add_dependencies(core_plugin_contract_tests contract_fault_plugins)
  target_compile_features(core_plugin_contract_tests PRIVATE cxx_std_17)
  target_link_libraries(core_plugin_contract_tests litesim3d_core::litesim3d_core GTest::gtest_main)
  ament_target_dependencies(core_plugin_contract_tests ament_index_cpp)
  gtest_discover_tests(core_plugin_contract_tests DISCOVERY_TIMEOUT 60)
  if(LITESIM_QUALITY_MODE STREQUAL "asan")
    # Never ship the deliberately faulty fixture in a normal/release build.
    find_package(pluginlib REQUIRED)
    add_library(leaky_test_plugin SHARED leaky_test_plugin.cpp)
    target_compile_features(leaky_test_plugin PRIVATE cxx_std_17)
    target_link_libraries(leaky_test_plugin litesim3d_plugin_api::litesim3d_plugin_api)
    ament_target_dependencies(leaky_test_plugin pluginlib)
    pluginlib_export_plugin_description_file(litesim3d_plugin_api leaky_test_plugin.xml)
    install(TARGETS leaky_test_plugin LIBRARY DESTINATION lib)
    add_executable(plugin_leak_probe plugin_leak_probe.cpp)
    target_compile_features(plugin_leak_probe PRIVATE cxx_std_17)
    target_link_libraries(plugin_leak_probe litesim3d_plugin_api::litesim3d_plugin_api)
  endif()
endif()
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
