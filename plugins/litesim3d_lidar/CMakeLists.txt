cmake_minimum_required(VERSION 3.20)
project(litesim3d_lidar VERSION 0.4.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(litesim3d_plugin_api REQUIRED)
find_package(pluginlib REQUIRED)
find_package(litesim3d_core REQUIRED)
add_library(${PROJECT_NAME} SHARED src/plugin.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_link_libraries(${PROJECT_NAME} litesim3d_plugin_api::litesim3d_plugin_api litesim3d_core::litesim3d_core)
ament_target_dependencies(${PROJECT_NAME} pluginlib)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(litesim3d_plugin_api plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
