cmake_minimum_required(VERSION 3.20)
project(litesim3d_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(litesim3d_core REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(litesim3d_runtime REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
set(DEPS rclcpp rcl rmw pluginlib geometry_msgs sensor_msgs nav_msgs rosgraph_msgs std_msgs std_srvs diagnostic_msgs visualization_msgs tf2_msgs litesim3d_interfaces)
find_package(tf2_ros REQUIRED)
foreach(dep ${DEPS})
  find_package(${dep} REQUIRED)
endforeach()
add_library(litesim3d_lidar_output_api INTERFACE)
target_include_directories(litesim3d_lidar_output_api INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(litesim3d_lidar_output_api INTERFACE litesim3d_core::litesim3d_core rclcpp::rclcpp)
# Record the installed ROS/launch implementation independently of the core hash.
set(bridge_sources src/joint_control.hpp src/robot_view.hpp src/simulator_backend.hpp src/world_backend.hpp src/cloud_tool_boundary.hpp src/lidar_outputs.cpp src/lidar_outputs.hpp include/litesim3d_ros2/lidar_output_plugin.hpp src/sim_node.cpp src/sim_world_node.cpp src/point_adapter.cpp
  src/point_adapter.hpp src/service_responses.hpp src/sim_ouster_adapter.cpp src/sim_cloud_probe.cpp
  launch/demo.launch.py launch/world.launch.py src/sim_session.cpp src/session_protocol.cpp src/session_protocol.hpp src/ros_client.cpp)
set(bridge_fingerprint "")
foreach(source ${bridge_sources})
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND bridge_fingerprint "${source}:${source_hash}\n")
endforeach()
string(SHA256 bridge_hash "${bridge_fingerprint}")
add_compile_definitions(LWS_BRIDGE_SHA256="${bridge_hash}")
include_directories(include)
add_executable(sim_node src/sim_node.cpp src/lidar_outputs.cpp)
# Use the library API only. Linking every tf2_ros export also loads the unused
# static-broadcaster component and its process-lifetime class_loader registry.
target_link_libraries(sim_node litesim3d_core::litesim3d_core tf2_ros::tf2_ros)
ament_target_dependencies(sim_node ${DEPS})
target_compile_options(sim_node PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sim_world_node src/sim_world_node.cpp src/lidar_outputs.cpp)
target_link_libraries(sim_world_node litesim3d_core::litesim3d_core tf2_ros::tf2_ros)
ament_target_dependencies(sim_world_node ${DEPS})
find_package(PCL REQUIRED COMPONENTS common filters)
add_executable(sim_ouster_adapter src/sim_ouster_adapter.cpp src/point_adapter.cpp)
ament_target_dependencies(sim_ouster_adapter rclcpp sensor_msgs std_msgs)
add_executable(sim_cloud_probe src/sim_cloud_probe.cpp)
ament_target_dependencies(sim_cloud_probe rclcpp sensor_msgs std_msgs)
target_include_directories(sim_cloud_probe SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(sim_cloud_probe ${PCL_LIBRARIES} nlohmann_json::nlohmann_json)
target_link_libraries(sim_ouster_adapter nlohmann_json::nlohmann_json)
if(SIM_COVERAGE)
  foreach(target sim_ouster_adapter sim_cloud_probe)
    target_compile_options(${target} PRIVATE -fprofile-instr-generate -fcoverage-mapping)
    target_link_options(${target} PRIVATE -fprofile-instr-generate)
  endforeach()
endif()
install(TARGETS sim_cloud_probe sim_ouster_adapter sim_world_node sim_node DESTINATION lib/${PROJECT_NAME})
add_executable(sim_session src/sim_session.cpp src/session_protocol.cpp)
target_link_libraries(sim_session litesim3d_runtime::litesim3d_runtime)
ament_target_dependencies(sim_session rclcpp rmw rcl_interfaces diagnostic_msgs geometry_msgs litesim3d_interfaces ament_index_cpp)
target_compile_options(sim_session PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS sim_session DESTINATION lib/${PROJECT_NAME})
add_executable(ros_client src/ros_client.cpp)
target_link_libraries(ros_client litesim3d_runtime::litesim3d_runtime)
ament_target_dependencies(ros_client rclcpp geometry_msgs nav_msgs sensor_msgs rosgraph_msgs)
install(TARGETS ros_client DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(TARGETS litesim3d_lidar_output_api EXPORT lidar_output_apiTargets)
ament_export_targets(lidar_output_apiTargets)
ament_export_include_directories(include)
ament_export_dependencies(litesim3d_core rclcpp)
