cmake_minimum_required(VERSION 3.20)
project(litesim_metal LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OGRENEXT REQUIRED IMPORTED_TARGET OGRE-Next=2.2.6)
set(METAL_SOURCE_DIR "" CACHE PATH "Verified Ogre-Next source cache")
set(metal "${METAL_SOURCE_DIR}/RenderSystems/Metal")
file(GLOB sources "${metal}/src/*.mm" "${metal}/src/Vao/*.mm" "${metal}/src/Windowing/OSX/*.mm")
add_library(RenderSystem_Metal SHARED ${sources})
set_target_properties(RenderSystem_Metal PROPERTIES PREFIX "" OUTPUT_NAME "RenderSystem_Metal")
target_compile_options(RenderSystem_Metal PRIVATE -fobjc-arc -Wno-deprecated-declarations -Wno-gnu-inline-cpp-without-extern)
target_compile_definitions(RenderSystem_Metal PRIVATE OGRE_METAL_EXPORTS)
target_include_directories(RenderSystem_Metal PRIVATE "${metal}/include" "${metal}/include/Windowing/OSX")
target_link_libraries(RenderSystem_Metal PkgConfig::OGRENEXT "-framework Metal" "-framework AppKit" "-framework QuartzCore")
