cmake_minimum_required(VERSION 3.20)
project(litesim3d_mid360 VERSION 0.4.0 LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(litesim3d_plugin_api REQUIRED)
find_package(litesim3d_core REQUIRED)
find_package(litesim3d_ros2 REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)

# Separate libraries: core sensor loading does not load ROS message type support.
add_library(${PROJECT_NAME} SHARED src/plugin.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_link_libraries(${PROJECT_NAME} litesim3d_plugin_api::litesim3d_plugin_api litesim3d_core::litesim3d_core)
ament_target_dependencies(${PROJECT_NAME} pluginlib)
pluginlib_export_plugin_description_file(litesim3d_plugin_api plugins.xml)

add_library(${PROJECT_NAME}_output SHARED src/livox_outputs.cpp)
target_compile_features(${PROJECT_NAME}_output PUBLIC cxx_std_17)
target_link_libraries(${PROJECT_NAME}_output litesim3d_ros2::litesim3d_lidar_output_api)
ament_target_dependencies(${PROJECT_NAME}_output pluginlib rclcpp sensor_msgs livox_ros_driver2)
pluginlib_export_plugin_description_file(litesim3d_ros2 output_plugins.xml)
foreach(target ${PROJECT_NAME} ${PROJECT_NAME}_output)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS ${PROJECT_NAME} ${PROJECT_NAME}_output LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  enable_testing()
  add_executable(mid360_plugin_tests test/test_plugin.cpp)
  target_link_libraries(mid360_plugin_tests litesim3d_core::litesim3d_core GTest::gtest_main)
  ament_target_dependencies(mid360_plugin_tests ament_index_cpp)
  add_executable(livox_outputs_unit test/test_livox_outputs_unit.cpp)
  target_compile_features(livox_outputs_unit PRIVATE cxx_std_17)
  target_link_libraries(livox_outputs_unit ${PROJECT_NAME}_output GTest::gtest_main)
  ament_target_dependencies(livox_outputs_unit rclcpp sensor_msgs livox_ros_driver2)
  include(GoogleTest)
  gtest_discover_tests(mid360_plugin_tests DISCOVERY_TIMEOUT 60 DISCOVERY_MODE PRE_TEST)
  gtest_discover_tests(livox_outputs_unit DISCOVERY_TIMEOUT 60 DISCOVERY_MODE PRE_TEST)
endif()
ament_package()
