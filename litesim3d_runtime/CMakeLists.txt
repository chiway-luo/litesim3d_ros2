cmake_minimum_required(VERSION 3.20)
project(litesim3d_runtime LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(${PROJECT_NAME} SHARED src/resources.cpp src/process.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} PUBLIC yaml-cpp nlohmann_json::nlohmann_json)
ament_target_dependencies(${PROJECT_NAME} PUBLIC ament_index_cpp)
install(TARGETS ${PROJECT_NAME} EXPORT runtimeTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(runtimeTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(ament_index_cpp yaml-cpp nlohmann_json)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(runtime_unit test/test_runtime.cpp)
  target_link_libraries(runtime_unit ${PROJECT_NAME})
  # The actual process implementation is compiled with translation-unit-local
  # deterministic POSIX boundaries; the subprocess suite remains integration-only.
  ament_add_gtest(runtime_process_unit test/test_process_unit.cpp)
  target_link_libraries(runtime_process_unit ${PROJECT_NAME})
  ament_add_gtest(runtime_process_integration test/test_process.cpp)
  target_link_libraries(runtime_process_integration ${PROJECT_NAME})
endif()
ament_package()
