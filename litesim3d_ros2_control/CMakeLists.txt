cmake_minimum_required(VERSION 3.20)
project(litesim3d_ros2_control LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
set(DEPS hardware_interface pluginlib rclcpp rclcpp_lifecycle litesim3d_interfaces)
foreach(dep ${DEPS})
  find_package(${dep} REQUIRED)
endforeach()
add_library(litesim3d_system SHARED src/system.cpp)
target_compile_features(litesim3d_system PUBLIC cxx_std_17)
target_compile_options(litesim3d_system PRIVATE -Wall -Wextra -Wpedantic)
# Humble's hardware_interface exports fake/mock plugin targets as well as the
# interface. Linking every export loads their registrations twice in a manager.
target_link_libraries(litesim3d_system hardware_interface::hardware_interface)
ament_target_dependencies(litesim3d_system pluginlib rclcpp rclcpp_lifecycle litesim3d_interfaces)
pluginlib_export_plugin_description_file(hardware_interface hardware_plugins.xml)
find_package(controller_interface REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(controller_manager REQUIRED)
add_executable(ros2_control_node src/ros2_control_node.cpp)
target_compile_features(ros2_control_node PRIVATE cxx_std_17)
ament_target_dependencies(ros2_control_node controller_manager rclcpp realtime_tools)
install(TARGETS ros2_control_node DESTINATION lib/${PROJECT_NAME})
find_package(Eigen3 REQUIRED)
add_library(litesim3d_omni_controller SHARED src/omni_controller.cpp src/body_controller.cpp)
target_compile_features(litesim3d_omni_controller PUBLIC cxx_std_17)
target_compile_options(litesim3d_omni_controller PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(litesim3d_omni_controller Eigen3::Eigen)
ament_target_dependencies(litesim3d_omni_controller controller_interface geometry_msgs realtime_tools pluginlib rclcpp)
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
install(TARGETS litesim3d_system litesim3d_omni_controller LIBRARY DESTINATION lib)
install(PROGRAMS scripts/spawn_when_ready.py DESTINATION lib/${PROJECT_NAME})
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(control_loop_sleep test/test_control_loop_sleep.cpp)
  target_include_directories(control_loop_sleep PRIVATE src)
  ament_target_dependencies(control_loop_sleep rclcpp)
  ament_add_gtest(body_configuration test/test_body_configuration.cpp)
  target_link_libraries(body_configuration hardware_interface::hardware_interface)
  ament_target_dependencies(body_configuration controller_interface pluginlib rclcpp)
  ament_add_gtest(omni_configuration test/test_omni_configuration.cpp)
  ament_target_dependencies(omni_configuration controller_interface pluginlib rclcpp)
  ament_add_pytest_test(control_integration test/test_integration.py TIMEOUT 180
    APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_pytest_test(control_startup test/test_spawner.py TIMEOUT 60
    APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
  ament_add_pytest_test(control_shutdown test/test_shutdown.py TIMEOUT 90
    APPEND_ENV PYTEST_DISABLE_PLUGIN_AUTOLOAD=1)
endif()
ament_package()
