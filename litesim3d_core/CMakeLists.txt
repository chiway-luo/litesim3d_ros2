cmake_minimum_required(VERSION 3.20)
include(cmake/version.cmake)
project(litesim3d_core VERSION ${LITESIM3D_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Eigen3 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(URDF REQUIRED IMPORTED_TARGET urdfdom)
find_package(assimp REQUIRED)
find_package(fcl REQUIRED)
find_package(yaml-cpp REQUIRED)
pkg_check_modules(TINYXML2 REQUIRED IMPORTED_TARGET tinyxml2)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(litesim3d_plugin_api REQUIRED)
option(SIM_COVERAGE "Enable Clang source coverage" OFF)
option(SIM_SANITIZE "Enable ASan and UBSan" OFF)
option(SIM_TSAN "Enable the separate thread sanitizer build" OFF)
if(SIM_SANITIZE AND SIM_TSAN)
  message(FATAL_ERROR "ThreadSanitizer must be built separately from ASan/UBSan")
endif()
add_library(litesim3d_core SHARED src/geometry.cpp src/model.cpp src/config.cpp src/simulator.cpp src/sensors.cpp src/terrain.cpp src/world.cpp src/collision_budget.cpp src/plugins.cpp src/sensor_plugin.cpp)
target_compile_features(litesim3d_core PUBLIC cxx_std_17)
set(core_fingerprints "")
foreach(source include/litesim3d/sim.hpp include/litesim3d/rate.hpp include/litesim3d/sensor_plugin.hpp src/geometry.cpp src/model.cpp src/config.cpp src/simulator.cpp src/sensors.cpp src/terrain.cpp src/world.cpp src/collision_budget.cpp src/plugins.cpp src/sensor_plugin.cpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" fingerprint)
  string(APPEND core_fingerprints "${source}:${fingerprint}\n")
endforeach()
string(SHA256 core_hash "${core_fingerprints}")
target_compile_definitions(litesim3d_core PRIVATE LWS_VERSION="${PROJECT_VERSION}" LWS_CORE_SHA256="${core_hash}" LWS_URDFDOM_VERSION="${URDF_VERSION}")
target_include_directories(litesim3d_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(litesim3d_core PUBLIC Eigen3::Eigen fcl yaml-cpp nlohmann_json::nlohmann_json litesim3d_plugin_api::litesim3d_plugin_api PRIVATE PkgConfig::URDF assimp::assimp PkgConfig::TINYXML2 OpenSSL::Crypto)
target_compile_options(litesim3d_core PRIVATE -Wall -Wextra -Wpedantic)
if(SIM_COVERAGE)
  target_compile_options(litesim3d_core PUBLIC -fprofile-instr-generate -fcoverage-mapping)
  target_link_options(litesim3d_core PUBLIC -fprofile-instr-generate)
endif()
if(SIM_SANITIZE)
  target_compile_options(litesim3d_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(litesim3d_core PUBLIC -fsanitize=address,undefined)
endif()
if(SIM_TSAN)
  target_compile_options(litesim3d_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(litesim3d_core PUBLIC -fsanitize=thread)
endif()
add_executable(sim_headless src/headless.cpp)
target_link_libraries(sim_headless litesim3d_core)
add_executable(sim_load_benchmark src/load_benchmark.cpp)
target_link_libraries(sim_load_benchmark litesim3d_core)
include(CTest)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(sim_core_tests test/test_core.cpp test/test_m2.cpp test/test_m3.cpp test/test_lidar_patterns.cpp test/test_mid360.cpp test/test_review_fixes.cpp test/test_sdf_budget_rate.cpp test/test_camera.cpp test/test_multicopter.cpp test/test_collision_odometry.cpp)
  target_link_libraries(sim_core_tests litesim3d_core GTest::gtest_main)
  target_compile_definitions(sim_core_tests PRIVATE SIM_RESOURCES="${CMAKE_CURRENT_SOURCE_DIR}/resources")
  add_executable(lifecycle_probe test/lifecycle_probe.cpp)
  target_link_libraries(lifecycle_probe litesim3d_core)
  include(GoogleTest)
  gtest_discover_tests(sim_core_tests DISCOVERY_TIMEOUT 60 DISCOVERY_MODE PRE_TEST)
endif()
install(TARGETS litesim3d_core sim_headless sim_load_benchmark EXPORT litesim3d_coreTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY resources/ DESTINATION share/${PROJECT_NAME})
install(FILES "${LITESIM3D_VERSION_FILE}" DESTINATION share/${PROJECT_NAME})
find_package(ament_cmake QUIET)
if(ament_cmake_FOUND AND CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  ament_export_targets(litesim3d_coreTargets HAS_LIBRARY_TARGET)
  ament_export_dependencies(Eigen3 fcl yaml-cpp nlohmann_json litesim3d_plugin_api)
  ament_package(CONFIG_EXTRAS "cmake/dependencies.cmake")
endif()
