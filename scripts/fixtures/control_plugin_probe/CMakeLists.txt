cmake_minimum_required(VERSION 3.20)
project(control_plugin_probe LANGUAGES CXX)
find_package(hardware_interface REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
add_library(control_probe_plugins SHARED plugins.cpp)
target_link_libraries(control_probe_plugins hardware_interface::hardware_interface controller_interface::controller_interface pluginlib::pluginlib)
add_executable(control_plugin_probe main.cpp)
target_link_libraries(control_plugin_probe hardware_interface::hardware_interface controller_interface::controller_interface pluginlib::pluginlib)
foreach(target control_probe_plugins control_plugin_probe)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE -O0 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_compile_options(${target} PRIVATE -fno-sanitize-recover=all)
  target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
endforeach()
add_executable(control_plugin_probe_stripped main.cpp)
target_compile_features(control_plugin_probe_stripped PRIVATE cxx_std_17)
target_compile_options(control_plugin_probe_stripped PRIVATE -O2)
target_link_options(control_plugin_probe_stripped PRIVATE -Wl,-s)
target_link_libraries(control_plugin_probe_stripped hardware_interface::hardware_interface controller_interface::controller_interface pluginlib::pluginlib)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  target_link_options(control_plugin_probe PRIVATE -no-pie)
endif()
install(TARGETS control_probe_plugins LIBRARY DESTINATION lib)
install(FILES plugins.xml DESTINATION share/control_plugin_probe)
install(FILES package.xml.in DESTINATION share/control_plugin_probe RENAME package.xml)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/marker" "")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/export" "share/control_plugin_probe/plugins.xml")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/marker" DESTINATION share/ament_index/resource_index/packages RENAME control_plugin_probe)
foreach(base hardware_interface controller_interface)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/export" DESTINATION "share/ament_index/resource_index/${base}__pluginlib__plugin" RENAME control_plugin_probe)
endforeach()
